### 5. 数据获取

#### 从positions表获取数据
```typescript
const fetchPositions = async () => {
  if (!userInfo || !userInfo.id) return;
  
  try {
    const { data, error } = await supabase
      .from('positions')
      .select(`
        *, 
        products (
          id,
          fund_number,
          name_cn,
          name_en,
          net_asset_value
        )
      `)
      .eq('user_id', userInfo.id);
    
    if (error) throw error;
    
    setPositions(data || []);
  } catch (error) {
    console.error('获取持仓数据失败:', error);
  } finally {
    setLoading(false);
    setRefreshing(false);
  }
};
```
